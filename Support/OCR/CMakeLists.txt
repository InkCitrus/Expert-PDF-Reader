cmake_minimum_required(VERSION 3.20)
project(ReaderOCR LANGUAGES C CXX)
set(PPOCR_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(PPOCR_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(PPOCR_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(PPOCR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(PPOCR_ENABLE_VULKAN OFF CACHE BOOL "" FORCE)
set(PPOCR_ENABLE_METAL OFF CACHE BOOL "" FORCE)
set(PPOCR_INDRA_METAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${RAPIDOCR_SOURCE}" rapidocr)
add_library(ReaderOCR SHARED Bridge.cpp)
target_link_libraries(ReaderOCR PRIVATE ppocr)
target_compile_features(ReaderOCR PRIVATE cxx_std_20)
set_target_properties(ReaderOCR PROPERTIES INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR YES)
